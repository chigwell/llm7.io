function ListItem({
  title,

  href,
  icon,
  external = false,
  ...props

  href: string;

  external?: boolean;
}) {
  return (
    <li {...props}>
      <Link
        href={href}
        target={external ? "_blank" : undefined}
        rel={external ? "noopener noreferrer" : undefined}

            {icon}

    </li>
  );
}
